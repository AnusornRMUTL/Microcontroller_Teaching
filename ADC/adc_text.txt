#include <xc.h>
#define _XTAL_FREQ 20000000
#pragma config FOSC = HS, WDTE = OFF, LVP = OFF

void main(void)
{  
	ANSEL = 0b00000010;
    ANSELH = 0x00;
    TRISB = 0x00;  
    PORTB = 0x00;
    TRISC = 0b10000000;
    PORTC = 0x00;
    
    // USART module setting.
    TXSTAbits.SYNC = 0;   
    TXSTAbits.BRGH = 0;  
    BAUDCTLbits.BRG16 = 1;      
    // Baud rate = 9600 bps
    SPBRGH = 0x00;
    SPBRG = 0x81;     
    
    RCSTAbits.SPEN = 1;
    RCSTAbits.CREN = 1;
    TXSTAbits.TXEN = 1;    
    
    // ADC Module setting.
    INTCON = 0; 
    ADCON0 = 0b11000001; 
    ADCON1 = 0x00;
    ADCON0bits.CHS = 1;

    while (1)
    {
        ADCON0bits.GO_DONE = 1;
        while (ADCON0bits.GO_DONE);
              
        TXREG = (ADRESH/100) + 48;
        while(!TRMT);
        
        ADRESH = ADRESH - (ADRESH/100)*100;
        TXREG = (ADRESH/10) + 48;
        while(!TRMT);
        
        ADRESH = ADRESH - (ADRESH/10)*10;
        TXREG = ADRESH + 48;

        TXREG = 0x0D;   while(!TRMT);

     }
}
